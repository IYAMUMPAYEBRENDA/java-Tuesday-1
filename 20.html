let users = [ { id: 1, name: 'John' }, { id: 2, name: 'Jane' } ]; // GET all users app.get('/api/users', (req, res) => res.json(users)); // GET user by ID app.get('/api/users/:id', (req, res) => { const user = users.find(u => u.id === parseInt(req.params.id)); res.json(user); }); // POST new user app.post('/api/users', (req, res) => { const user = { id: users.length + 1, ...req.body }; users.push(user); res.status(201).json(user); });
